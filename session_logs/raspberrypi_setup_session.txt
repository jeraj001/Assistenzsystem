def initialize_pan_reference(
    current_dataset_pan_angle,
):
    current_physical_pan_angle = (
        current_dataset_pan_angle + 90
    )

    if not (
        0
        <= current_physical_pan_angle
        <= 180
    ):
        raise ValueError(
            f"Invalid current dataset "
            f"pan angle: "
            f"{current_dataset_pan_angle}"
        )

    ttt.current_angles[0] = (
        current_physical_pan_angle
    )

    ttt.pwm.setRotationAngle(
        0,
        current_physical_pan_angle,
    )

    print(
        f"Pan software reference "
        f"initialized: "
        f"dataset="
        f"{current_dataset_pan_angle:.2f}, "
        f"control="
        f"{current_physical_pan_angle:.2f}",
        flush=True,
    )

    time.sleep(0.5)

def move_motor(
    motor,

    clockwise,
):
    print(
        f"Moving motor by "

def verify_angle(
    actual_angle,
    expected_angle,
    angle_name,
):
    difference = abs(
        actual_angle - expected_angle
    )

    if difference > ANGLE_TOLERANCE:
        raise RuntimeError(
            f"{angle_name} angle mismatch: "
            f"expected {expected_angle:.2f}, "
            f"got {actual_angle:.2f}"
        )

def move_pan_to_dataset_angle(
    dataset_pan_angle,

    target_stepper1_angle,
    target_stepper2_angle,

    (
        current_stepper1_angle,
        current_stepper2_angle,
    ) = get_dataset_angles()

    stepper1_difference = (
        target_stepper1_angle
        - current_stepper1_angle

    if abs(stepper1_difference) > (
        ANGLE_TOLERANCE

        move_motor(
            motor=stepper,
            angle=abs(
                stepper1_difference

            clockwise=(

    stepper2_difference = (
        target_stepper2_angle
        - current_stepper2_angle
    )

    if abs(stepper2_difference) > (
        ANGLE_TOLERANCE
    ):
        move_motor(
            motor=stepper2,
            angle=abs(
                stepper2_difference
            ),
            clockwise=(
                stepper2_difference > 0
            ),
        )

    (
        actual_stepper1_angle,
        actual_stepper2_angle,
    ) = get_dataset_angles()

    verify_angle(
        actual_angle=actual_stepper1_angle,
        expected_angle=target_stepper1_angle,
        angle_name="Stepper 1 target",
    )

    verify_angle(
        actual_angle=actual_stepper2_angle,
        expected_angle=target_stepper2_angle,
        angle_name="Stepper 2 target",
    )

def build_recording_conditions():
    return [
        (
            pan_angle,
            stepper1_angle,
            stepper2_angle,
        )
        for pan_angle in PAN_TARGETS
        for stepper1_angle in STEPPER_TARGETS
        for stepper2_angle in STEPPER_TARGETS
    ]

def run_from_condition(start_condition):
    conditions = build_recording_conditions()

    if start_condition not in conditions:
        raise ValueError(
            f"Invalid start condition: "
            f"{start_condition}"
        )

    start_index = conditions.index(
        start_condition
    )

    current_pan_dataset_angle = (
        CURRENT_PAN_DATASET_ANGLE
    )

    for (
        target_pan_angle,
        target_stepper1_angle,
        target_stepper2_angle,
    ) in conditions[start_index:]:
        move_steppers_to(
            target_stepper1_angle=(
                target_stepper1_angle
            ),
            target_stepper2_angle=(
                target_stepper2_angle
            ),
        )

        if (
            target_pan_angle
            != current_pan_dataset_angle
        ):
            physical_pan_angle = (
                move_pan_to_dataset_angle(
                    target_pan_angle
                )
            )

            current_pan_dataset_angle = (
                target_pan_angle
            )
        else:
            physical_pan_angle = (
                target_pan_angle + 90
            )

        record_current_condition(
            physical_pan_angle=(
                physical_pan_angle
            ),
            expected_stepper1_angle=(
                target_stepper1_angle
            ),
            expected_stepper2_angle=(
                target_stepper2_angle
            ),
        )

def main():

from gpiozero import OutputDevice
import time
from time import sleep

STEP_PIN = 16
DIR_PIN = 26

PULSES_PER_REV = 1600
DEFAULT_STEP_DELAY = 0.05
PULSE_WIDTH = 0.0005

step = OutputDevice(STEP_PIN)
direction = OutputDevice(DIR_PIN)
current_position_steps = 0


def move_steps(steps, clockwise=True, step_delay=DEFAULT_STEP_DELAY):
    global current_position_steps

    direction.value = 1 if clockwise else 0
    sleep(0.1)

    for _ in range(steps):
        step.on()
        sleep(PULSE_WIDTH)
        step.off()
        sleep(step_delay)

        if clockwise:
            current_position_steps += 1

from gpiozero import OutputDevice
import time
from time import sleep

STEP_PIN = 16
DIR_PIN = 26

PULSES_PER_REV = 1600
DEFAULT_STEP_DELAY = 0.05
PULSE_WIDTH = 0.0005

step = OutputDevice(STEP_PIN)
direction = OutputDevice(DIR_PIN)
current_position_steps = 0


def move_steps(steps, clockwise=True, step_delay=DEFAULT_STEP_DELAY):
    global current_position_steps

    direction.value = 1 if clockwise else 0
    sleep(0.1)

    for _ in range(steps):
        step.on()
        sleep(PULSE_WIDTH)
        step.off()
        sleep(step_delay)

        if clockwise:
            current_position_steps += 1
        else:
            current_position_steps -= 1


def move_angle(angle_deg, clockwise=True, step_delay=DEFAULT_STEP_DELAY):
    steps = round(PULSES_PER_REV * angle_deg / 360)

    print(f"{angle_deg} degrees = {steps} pulses", flush=True)

    move_steps(
        steps=steps,
        clockwise=clockwise,
        step_delay=step_delay,
    )

def get_current_angle():
    return current_position_steps * 360 / PULSES_PER_REV

def set_current_angle(angle):
    global current_position_steps

    current_position_steps = round(
        float(angle)
        / 360.0
        * PULSES_PER_REV
    )

        try:
                time.sleep(3)
                move_angle(15, clockwise=True)
                time.sleep(3)
                move_angle(15, clockwise=False)

if __name__ == "__main__":
